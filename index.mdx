Falconer is a fully end-to-end knowledge platform that provides humans and agents with auto-updating documentation.

<Card icon="plug" href="/set-up/connect-sources" cta="Connect sources">
  ## Connect sources

  Connect tools like GitHub, Linear, Jira, Slack, Notion, Google Docs, and Confluence to connect context or sync with your Falconer docs.
</Card>

<Card icon="heart-rate-monitor" href="/update/find-and-fix-outdated-docs/" cta="Find outdated docs">
  ## Find issues

  Knowledge Health reads through all of your docs and connected sources to find the issues that degrade quality, like gaps, contradictions, and code drift.
</Card>

<Card icon="refresh" href="/update/auto-update-docs/" cta="Set up auto-update">
  ## Auto-update

  Set your docs to auto-update whenever any related code changes impact them. Your docs listen for changes and can either make the changes automatically or ask for your review.
</Card>

<Card icon="pencil" href="/write/write-a-doc/" cta="Write a doc">
  ## Write docs

  With your sources connected, Falconer writes expert-level docs from a prompt in the UI, or from Claude Code, Cursor, Codex, and Slack. Draft guides, runbooks, changelogs, and diagrams with citations to the actual sources so you can verify.
</Card>

    };

    if (reduce || !('IntersectionObserver' in window)) {
      cards.forEach(c => c.classList.add('is-in'));
      strips.forEach(s => paint(s, 3));
      addEventListener('resize', () => strips.forEach(s => paint(s, 3)));
    } else {
      const io = new IntersectionObserver(entries => {
        entries.forEach(e => { if (e.isIntersecting) { e.target.classList.add('is-in'); io.unobserve(e.target); } });
      }, { threshold: 0.18, rootMargin: '0px 0px -8% 0px' });
      cards.forEach(c => io.observe(c));

      // Only animate strips on screen; each fades in gently the first time.
      const vis = new IntersectionObserver(entries => {
        entries.forEach(e => { const s = strips.find(x => x.canvas === e.target); if (s) s.visible = e.isIntersecting; });
      });
      strips.forEach(s => vis.observe(s.canvas));
      let last = performance.now(), clock = 0;
      const loop = (now) => {
        const dt = Math.min(.05, (now - last) / 1000); last = now; clock += dt;
        strips.forEach(s => {
          if (!s.visible) return;
          s.fade = Math.min(1, s.fade + dt / .8);
          paint(s, clock, easeOut(s.fade));
        });
        requestAnimationFrame(loop);
      };
      requestAnimationFrame(loop);
    }


    if (!fine || reduce) return;
    cards.forEach(card => {
      card.addEventListener('pointermove', e => {
        const r = card.getBoundingClientRect();
        const x = (e.clientX - r.left) / r.width, y = (e.clientY - r.top) / r.height;
        card.classList.add('is-tilting');
        card.style.setProperty('--mx', `${x * 100}%`);
        card.style.setProperty('--my', `${y * 100}%`);
        card.style.setProperty('--ry', `${(x - 0.5) * 6}deg`);
        card.style.setProperty('--rx', `${(0.5 - y) * 6}deg`);
      });
      card.addEventListener('pointerleave', () => {
        card.classList.remove('is-tilting');
        card.style.setProperty('--rx', '0deg');
        card.style.setProperty('--ry', '0deg');
      });
    });
  };
  document.readyState === 'loading' ? document.addEventListener('DOMContentLoaded', init) : init();
})();
