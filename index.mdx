const fills = [];

function part(group, geo, pos=[0,0,0], rot=[0,0,0], opt={}) {
  const base = opt.accent ? [TINT, ACC] : [PAPER, INK];
  const fill = new THREE.Mesh(geo, new THREE.MeshBasicMaterial({ color:base[0].clone(), transparent:true, opacity:opt.ghost ? 0 : 1,
    depthWrite:!opt.ghost, polygonOffset:true, polygonOffsetFactor:1, polygonOffsetUnits:1 }));
  const line = new THREE.LineSegments(new THREE.EdgesGeometry(geo, opt.th ?? 1), opt.ghost
    ? new THREE.LineDashedMaterial({ color:base[1].clone(), dashSize:.3, gapSize:.22, transparent:true })
    : new THREE.LineBasicMaterial({ color:base[1].clone(), transparent:true }));
  if (opt.ghost) line.computeLineDistances();
  fill.userData = { base, ghost:!!opt.ghost, key: opt.key ?? KEY, at: opt.at ?? AT };
  fill.add(line); fill.position.set(...pos); fill.rotation.set(...rot);
  group.add(fill); fills.push(fill);
  return fill;
}
const V = (x, y, z) => new THREE.Vector3(x, y, z);
const box = (w, h, d) => new THREE.BoxGeometry(w, h, d);

function rod(group, a, b, r=.1) {
  const A = V(...a), B = V(...b);

  f.quaternion.setFromUnitVectors(V(0,1,0), B.clone().sub(A).normalize());
  return f;
}

function spaceSections() {
  const on = wide.matches && content !== document.body;
  const set = (el, p, v) => el.style.setProperty(p, on ? v : '', on ? 'important' : '');
  // one short vertical dashed line, start-aligned, in the gap above each section heading

  content.querySelectorAll('h2').forEach(h => {
    const el = h.closest('.sl-heading-wrapper') || h;
    set(el, 'margin-top', '1rem');
    set(el, 'padding-top', '26vh');
    set(el, 'background-image', dash);
    set(el, 'background-repeat', 'no-repeat');
    set(el, 'background-size', '1.5px calc(26vh - 4.5rem)');
    set(el, 'background-position', '0 2rem');
  });
  const last = content.lastElementChild;
  if (last) set(last, 'margin-bottom', '16vh');
  set(content, 'position', 'relative');
  set(content, 'isolation', 'isolate');
  // Align the header's tabs and search with the content column. The header grid (FalconerHeader.astro)
  // subtracts 3 x --__toc-width from the free space and splits it in half, so a third of the space
  // reserved for the canvas shifts its middle column by the same amount the content shifts.
  document.documentElement.style.setProperty('--__toc-width', on ? 'calc(min(36vw, 34rem) / 3)' : '');
}
