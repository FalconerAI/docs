// Static state for the editor preview and small screens: the finished stack.
function staticStack() {
  sheets.forEach((el, i) => { const top = i === sheets.length - 1;
    el.style.transform = `translateZ(${i * GAP}px)`; el.style.opacity = 1;
    [baseOf(el), ...topsOf(el)].forEach(f => f.style.opacity = top ? 1 : .35);
    shadowsOf(el).forEach(sh => sh.style.opacity = top ? SHADOW : 0);
    colsOf(el).forEach(c => c.style.setProperty('--h', top ? colH(c) : '0px')); });
  sheets.at(-1).classList.add('on');
}

    set(el, 'margin-top', '1rem');

    set(el, 'background-image', dash);
    set(el, 'background-repeat', 'no-repeat');
    set(el, 'background-size', '1.5px calc(10vh - 2.5rem)');
    set(el, 'background-position', '0 1.25rem');
  });
  // a little room at the end; the last anchor is limited to the page's scroll range
  const last = content.lastElementChild;
  if (last) set(last, 'margin-bottom', '12vh');
  set(content, 'position', 'relative');
  set(content, 'isolation', 'isolate');

  // subtracts 3 x --__toc-width from the free space and splits it in half, so a third of the space
  // reserved for the stage shifts its middle column by the same amount the content shifts.

}
