Falconer is a fully end-to-end knowledge platform that provides humans and agents with auto-updating documentation.

Connect your code and tools, catch docs that have drifted, and keep every page current as your product changes.

  // The Falconer mark as a static blueprint: outlined steps, a guide diagonal, and one solid block.
  const ART = `<svg viewBox="-4 -4.7 35.2 35.2" fill="none" aria-hidden="true">
    <rect x="-4" y="-4.7" width="35.2" height="35.2" stroke="currentColor" stroke-opacity=".35" vector-effect="non-scaling-stroke"/>

      <path class="fx-step" style="--i:1" vector-effect="non-scaling-stroke" d="M10.0041 8.5956L8.28362 10.316V18.9166H16.8842L18.6047 17.1962H10.0057V8.59726L10.0041 8.5956Z"/>
      <path class="fx-step" style="--i:2" vector-effect="non-scaling-stroke" d="M18.6055 13.7574H13.4425V5.15511L11.722 6.87556V15.4745H20.321L22.0414 13.7541H18.6038L18.6055 13.7574Z"/>
      <path class="fx-step" style="--i:3" vector-effect="non-scaling-stroke" d="M22.0459 10.3166H16.8829V1.71602L15.1641 3.43647V12.0371H23.7631L25.4819 10.3166H22.0459Z"/>
      <rect class="fx-step" style="--i:4" vector-effect="non-scaling-stroke" x="18.597" y="0" width="8.599" height="8.599"/>
    </g>
  </svg>`;

  const init = () => {
    const content = document.querySelector('.sl-markdown-content');
    const cards = content ? [...content.querySelectorAll(':scope > .docsite-card.docsite-card-linked')] : [];
    if (!cards.length || content.querySelector('.fx-track')) return;

    const track = document.createElement('div');
    track.className = 'fx-track';
    cards[0].before(track);
    track.append(...cards);

    // Everything before the cards becomes the hero copy, led by the page title.
    const copy = document.createElement('div');
    copy.className = 'fx-hero-copy';
    const h1 = document.querySelector('.page-title-row h1');
    if (h1) copy.append(h1);
    while (content.firstElementChild !== track) copy.append(content.firstElementChild);
    copy.querySelectorAll('p').forEach(p => {
      const a = p.querySelector('a[href*="/quickstart"]');
      if (a && p.textContent.trim() === a.textContent.trim()) p.classList.add('fx-cta');
    });

    const art = document.createElement('div');
    art.className = 'fx-art';
    art.innerHTML = ART;

    const hero = document.createElement('div');
    hero.className = 'fx-hero';
    hero.append(copy, art);
    content.prepend(hero);

    const more = track.nextElementSibling;
    if (more && more.classList.contains('docsite-card')) more.classList.add('fx-more');

    // Hover glow and tilt, as in the previous card design.
    cards.forEach(c => c.insertAdjacentHTML('afterbegin', '<span class="fx-glow" aria-hidden="true"></span>'));
    const fine = matchMedia('(hover: hover) and (pointer: fine) and (min-width: 50em)').matches;
    const reduce = matchMedia('(prefers-reduced-motion: reduce)').matches;
    if (!fine || reduce) return;
    cards.forEach(card => {
      card.addEventListener('pointermove', e => {
        const r = card.getBoundingClientRect();
        const x = (e.clientX - r.left) / r.width, y = (e.clientY - r.top) / r.height;
        card.classList.add('is-tilting');
        card.style.setProperty('--mx', `${x * 100}%`);
        card.style.setProperty('--my', `${y * 100}%`);
        card.style.setProperty('--ry', `${(x - 0.5) * 6}deg`);
        card.style.setProperty('--rx', `${(0.5 - y) * 6}deg`);
      });
      card.addEventListener('pointerleave', () => {
        card.classList.remove('is-tilting');
        card.style.setProperty('--rx', '0deg');
        card.style.setProperty('--ry', '0deg');
      });
    });
  };
  document.readyState === 'loading' ? document.addEventListener('DOMContentLoaded', init) : init();
})();
</script>
